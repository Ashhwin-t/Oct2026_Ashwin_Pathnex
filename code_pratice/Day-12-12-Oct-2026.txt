Day 12 - Scaling application and Automation

Ansible - Setup Auto-Scaling Group
- name: Setup Auto Scalling Group
  hosts: all
  become: yes
  tasks:
	- name: Create Launch Configuration
	  ecw_asg:
	    name: pathnex-asg
		region: us-east-1
		launch_config_name: pathnex-launch-config
		min_size: 1
		max_size: 3
		desired_capacity: 2
		vpc_zone_identifier: subnet-12345678

Kubernetes - Horizontal Pod Autoscalar with Load balancer

apiVersion: app/v1
kind: Deployment
Metadata:
  name: pathnex-web-app
spec:
  replicas: 2
  selector:
    matchLabels:
	  app: pathnex-web-app
	template:
	  metadata:
	    labels:
		  app: pathnex-web-app
	  spec:
	    containers:
		- name: web
		  immage: nginx
		  ports:
		    - containerPort: 80
			
---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-web-app-hpa
  spec:
    scaleTargetRef:
	  apiVersion: apps/v1
	  kind: Deployment
	  name: pathnex-web-app
	minReplicas: 1
	maxReplicas: 5
	targetCPUUtilizationPercentage: 80
